  const codeBlocks = [];

    const code = pre.querySelector('code');

}

async function sendToServer(codeBlock) {
  try {
    const response = await fetch('http://localhost:8765/copy', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(codeBlock)
    });
    
    if (!response.ok) throw new Error('Failed to send');
    
    // Also copy to clipboard as fallback
    await navigator.clipboard.writeText(codeBlock.code);
  } catch (error) {
    console.error('Error sending to server:', error);
    // Fallback to clipboard only
    await navigator.clipboard.writeText(codeBlock.code);
  }
}

#!/usr/bin/env python3
import argparse
import pyperclip
import time
import json
import os
from pathlib import Path
import subprocess
import sys

class DeepSeekCopier:
    def __init__(self):
        self.config_dir = Path.home() / '.deepseek-copier'
        self.config_file = self.config_dir / 'config.json'
        self.history_file = self.config_dir / 'history.json'
        self.load_config()
    
    def load_config(self):
        self.config_dir.mkdir(exist_ok=True)
        if self.config_file.exists():
            self.config = json.loads(self.config_file.read_text())
        else:
            self.config = {
                'default_ide': 'vscode',
                'auto_copy': True,
                'watch_clipboard': False
            }
    
    def copy_to_ide(self, text, language='text', filename=None):
        """Copy text to IDE"""
        # Copy to clipboard
        pyperclip.copy(text)
        
        # Save to history
        self.save_history(text, language)
        
        # Try to open in IDE
        if filename:
            self.open_in_ide(text, language, filename)
        
        print(f"✅ Copied {len(text)} characters to clipboard")
        return True
    
    def save_history(self, text, language):
        """Save copy history"""
        history = []
        if self.history_file.exists():
            history = json.loads(self.history_file.read_text())
        
        history.append({
            'timestamp': time.time(),
            'language': language,
            'length': len(text),
            'content': text[:1000]  # Save first 1000 chars
        })
        
        # Keep last 100 entries
        history = history[-100:]
        self.history_file.write_text(json.dumps(history, indent=2))
    
    def open_in_ide(self, text, language, filename):
        """Open file in IDE"""
        # Create temp file
        temp_dir = self.config_dir / 'temp'
        temp_dir.mkdir(exist_ok=True)
        temp_file = temp_dir / filename
        
        # Write content
        temp_file.write_text(text)
        
        # Open in IDE based on config
        ide = self.config.get('default_ide', 'vscode')
        
        if ide == 'vscode':
            subprocess.Popen(['code', str(temp_file)])
        elif ide == 'sublime':
            subprocess.Popen(['subl', str(temp_file)])
        elif ide == 'atom':
            subprocess.Popen(['atom', str(temp_file)])
        else:
            print(f"⚠️ Unknown IDE: {ide}")
    
    def watch_clipboard(self):
        """Watch clipboard for changes"""
        print("👀 Watching clipboard... Press Ctrl+C to stop")
        last_content = ''
        
        try:
            while True:
                current = pyperclip.paste()
                if current != last_content:
                    print(f"📋 Clipboard changed: {len(current)} chars")
                    self.save_history(current, 'text')
                    last_content = current
                time.sleep(0.5)
        except KeyboardInterrupt:
            print("\n👋 Stopped watching")

def main():
    parser = argparse.ArgumentParser(description='DeepSeek to IDE Copier')
    parser.add_argument('--text', '-t', help='Text to copy')
    parser.add_argument('--file', '-f', help='File to copy from')
    parser.add_argument('--language', '-l', default='text', help='Code language')
    parser.add_argument('--filename', help='Filename for IDE')
    parser.add_argument('--watch', '-w', action='store_true', help='Watch clipboard')
    parser.add_argument('--ide', help='IDE to use (vscode, sublime, atom)')
    
    args = parser.parse_args()
    copier = DeepSeekCopier()
    
    if args.ide:
        copier.config['default_ide'] = args.ide
        copier.config_file.write_text(json.dumps(copier.config, indent=2))
        print(f"✅ Default IDE set to: {args.ide}")
    
    if args.watch:
        copier.watch_clipboard()
    elif args.text:
        copier.copy_to_ide(args.text, args.language, args.filename)
    elif args.file:
        text = Path(args.file).read_text()
        language = Path(args.file).suffix[1:] if Path(args.file).suffix else 'text'
        copier.copy_to_ide(text, language, args.filename or Path(args.file).name)
    else:
        # Read from stdin
        text = sys.stdin.read()
        if text:
            copier.copy_to_ide(text, args.language, args.filename)
        else:
            parser.print_help()

if __name__ == '__main__':
    main()

const vscode = require('vscode');
const http = require('http');
const WebSocket = require('ws');

let server;
let wss;
let statusBarItem;

function activate(context) {
    console.log('DeepSeek Copier is active');
    
    // Create status bar item
    statusBarItem = vscode.window.createStatusBarItem(
        vscode.StatusBarAlignment.Right, 100
    );
    statusBarItem.text = "$(cloud-download) DeepSeek";
    statusBarItem.command = 'deepseek.startServer';
    statusBarItem.show();
    
    // Register commands
    let startServer = vscode.commands.registerCommand(
        'deepseek.startServer', startServerCommand
    );
    
    context.subscriptions.push(startServer);
    context.subscriptions.push(statusBarItem);
    
    // Auto-start server
    startServerCommand();
}

function startServerCommand() {
    const config = vscode.workspace.getConfiguration('deepseek');
    const port = config.get('serverPort', 8765);
    
    if (server) {
        vscode.window.showInformationMessage('DeepSeek server already running');
        return;
    }
    
    // Create HTTP server for basic requests
    server = http.createServer((req, res) => {
        if (req.method === 'POST' && req.url === '/copy') {
            let body = '';
            req.on('data', chunk => body += chunk);
            req.on('end', () => {
                try {
                    const data = JSON.parse(body);
                    insertCode(data.code, data.language);
                    
                    res.writeHead(200, {'Content-Type': 'application/json'});
                    res.end(JSON.stringify({status: 'success'}));
                } catch (error) {
                    res.writeHead(500, {'Content-Type': 'application/json'});
                    res.end(JSON.stringify({status: 'error', message: error.message}));
                }
            });
        } else {
            res.writeHead(404);
            res.end();
        }
    });
    
    // Create WebSocket server for real-time updates
    wss = new WebSocket.Server({ server });
    
    wss.on('connection', (ws) => {
        ws.on('message', (message) => {
            try {
                const data = JSON.parse(message);
                if (data.type === 'code') {
                    insertCode(data.code, data.language);
                }
            } catch (error) {
                console.error('Error processing message:', error);
            }
        });
    });
    
    server.listen(port, () => {
        statusBarItem.text = `$(check) DeepSeek:${port}`;
        vscode.window.showInformationMessage(
            `DeepSeek server started on port ${port}`
        );
    });
}

function insertCode(code, language) {
    const editor = vscode.window.activeTextEditor;
    
    if (!editor) {
        vscode.window.showWarningMessage('No active editor');
        return;
    }
    
    const config = vscode.workspace.getConfiguration('deepseek');
    const autoInsert = config.get('autoInsert', true);
    
    if (autoInsert) {
        editor.edit((editBuilder) => {
            const position = editor.selection.active;
            editBuilder.insert(position, code);
        }).then(() => {
            vscode.window.showInformationMessage('Code inserted from DeepSeek');
        });
    }
}

function deactivate() {
    if (server) {
        server.close();
    }
    if (wss) {
        wss.close();
    }
}

module.exports = {
    activate,
    deactivate
};

        // Check position or styling
        const style = window.getComputedStyle(element);

    def detect_ides(self) -> Dict[str, Dict]:
        """Detect installed IDEs"""
        ides = {}
        system = platform.system()
        
        # VS Code
        vscode_paths = [
            'code',
            '/Applications/Visual Studio Code.app/Contents/Resources/app/bin/code',
            'C:\\Program Files\\Microsoft VS Code\\Code.exe',
            'C:\\Users\\' + os.getenv('USERNAME', '') + '\\AppData\\Local\\Programs\\Microsoft VS Code\\Code.exe'
        ]
        for path in vscode_paths:
            if self.check_command_exists(path):
                ides['vscode'] = {
                    'name': 'Visual Studio Code',
                    'command': path,
                    'type': 'vscode',
                    'extensions_dir': Path.home() / '.vscode' / 'extensions'
                }
                break
        
        # JetBrains IDEs
        jetbrains_ides = [
            ('pycharm', 'PyCharm'),
            ('intellij', 'IntelliJ IDEA'),
            ('webstorm', 'WebStorm'),
            ('phpstorm', 'PhpStorm'),
            ('goland', 'GoLand'),
            ('clion', 'CLion'),
            ('rider', 'Rider'),
            ('rubymine', 'RubyMine'),
            ('datagrip', 'DataGrip')
        ]
        
        if system == 'Darwin':  # macOS
            for ide_id, ide_name in jetbrains_ides:
                app_path = f'/Applications/{ide_name}.app'
                if os.path.exists(app_path):
                    ides[ide_id] = {
                        'name': ide_name,
                        'command': app_path,
                        'type': 'jetbrains',
                        'launcher': f'/usr/local/bin/{ide_id}'
                    }
        elif system == 'Windows':
            for ide_id, ide_name in jetbrains_ides:
                path = f'C:\\Program Files\\JetBrains\\{ide_name}'
                if os.path.exists(path):
                    ides[ide_id] = {
                        'name': ide_name,
                        'command': path,
                        'type': 'jetbrains'
                    }
        
        # Sublime Text
        sublime_paths = [
            'subl',
            '/Applications/Sublime Text.app/Contents/SharedSupport/bin/subl',
            'C:\\Program Files\\Sublime Text 3\\sublime_text.exe'
        ]
        for path in sublime_paths:
            if self.check_command_exists(path):
                ides['sublime'] = {
                    'name': 'Sublime Text',
                    'command': path,
                    'type': 'sublime'
                }
                break
        
        # Atom
        if self.check_command_exists('atom'):
            ides['atom'] = {
                'name': 'Atom',
                'command': 'atom',
                'type': 'atom'
            }

                ides['notepadpp'] = {

                }

    def check_command_exists(self, command: str) -> bool:
        """Check if command exists"""
        try:
            if os.path.exists(command):
                return True
            
            result = subprocess.run(
                ['which', command] if platform.system() != 'Windows' else ['where', command],
                capture_output=True,
                text=True
            )
            return result.returncode == 0
        except:
            return False
    
    def detect_active_ide(self) -> Optional[str]:
        """Detect which IDE is currently active/focused"""
        system = platform.system()
        
        if system == 'Darwin':  # macOS
            return self.detect_active_ide_macos()
        elif system == 'Windows':
            return self.detect_active_ide_windows()
        elif system == 'Linux':
            return self.detect_active_ide_linux()
        
        return None
    
    def detect_active_ide_macos(self) -> Optional[str]:
        """Detect active IDE on macOS"""
        try:
            # Use AppleScript to get active application
            script = 'tell application "System Events" to get name of first application process whose frontmost is true'
            result = subprocess.run(['osascript', '-e', script], capture_output=True, text=True)
            app_name = result.stdout.strip()
            
            # Map application names to IDE types
            ide_mapping = {
                'Visual Studio Code': 'vscode',
                'PyCharm': 'pycharm',
                'IntelliJ IDEA': 'intellij',
                'WebStorm': 'webstorm',
                'PhpStorm': 'phpstorm',
                'GoLand': 'goland',
                'CLion': 'clion',
                'Sublime Text': 'sublime',
                'Atom': 'atom'
            }
            
            return ide_mapping.get(app_name)
        except:
            return None
    
    def detect_active_ide_windows(self) -> Optional[str]:
        """Detect active IDE on Windows"""
        try:
            import win32gui
            import win32process
            
            # Get foreground window
            hwnd = win32gui.GetForegroundWindow()
            _, pid = win32process.GetWindowThreadProcessId(hwnd)
            
            # Get process name
            process = psutil.Process(pid)
            process_name = process.name().lower()
            
            # Map process names to IDE types
            ide_mapping = {
                'code.exe': 'vscode',
                'pycharm64.exe': 'pycharm',
                'idea64.exe': 'intellij',
                'webstorm64.exe': 'webstorm',
                'phpstorm64.exe': 'phpstorm',
                'goland64.exe': 'goland',
                'clion64.exe': 'clion',
                'sublime_text.exe': 'sublime',
                'atom.exe': 'atom',
                'notepad++.exe': 'notepadpp'
            }
            
            return ide_mapping.get(process_name)
        except:
            return None
    
    def detect_active_ide_linux(self) -> Optional[str]:
        """Detect active IDE on Linux"""
        try:
            # Use xdotool to get active window
            result = subprocess.run(
                ['xdotool', 'getactivewindow', 'getwindowname'],
                capture_output=True,
                text=True
            )
            window_title = result.stdout.strip().lower()
            
            # Check for IDE indicators in window title
            if 'visual studio code' in window_title or 'code' in window_title:
                return 'vscode'
            elif 'pycharm' in window_title:
                return 'pycharm'
            elif 'intellij' in window_title:
                return 'intellij'
            elif 'sublime' in window_title:
                return 'sublime'
            elif 'atom' in window_title:
                return 'atom'
            
            return None
        except:
            return None
    
    def get_ide_info(self, ide_id: str) -> Optional[Dict]:
        """Get IDE information"""
        return self.ides.get(ide_id)
    
    def list_available_ides(self) -> List[Dict]:
        """List all available IDEs"""
        return [
            {'id': ide_id, 'name': info['name'], 'type': info['type']}
            for ide_id, info in self.ides.items()
        ]

    def load_config(self) -> Dict:
        """Load configuration"""

        if config_file.exists():

        config_file.write_text(json.dumps(self.config, indent=2))
    
    def handle_copy(self, data: Dict) -> Dict:
        """Handle copy request"""
        result = {
            'status': 'success',
            'detected_ides': self.ide_detector.list_available_ides(),
            'active_ide': self.ide_detector.detect_active_ide()
        }
        
        # Determine target IDE
        target_ide = self.config.get('default_ide', 'auto')
        if target_ide == 'auto':
            target_ide = self.ide_detector.detect_active_ide()
        
        result['target_ide'] = target_ide
        
        # Handle different data types
        if data.get('type') == 'full_chat':
            self.handle_full_chat(data, target_ide)
        elif data.get('type') == 'message':
            self.handle_message(data, target_ide)
        elif data.get('type') == 'all_code':
            self.handle_all_code(data, target_ide)
        elif data.get('type') == 'markdown':
            self.handle_markdown(data, target_ide)
        else:
            self.handle_single_code(data, target_ide)
        
        # Save to history
        if self.config.get('save_history', True):
            self.save_history(data)
        
        return result
    
    def handle_full_chat(self, data: Dict, target_ide: str):

        # Create markdown file
        markdown = self.convert_chat_to_markdown(data)

        temp_dir = Path.home() / '.deepseek-copier' / 'temp'

        # Copy to clipboard
        pyperclip.copy(markdown)
    
    def handle_message(self, data: Dict, target_ide: str):
        """Handle single message"""
        content = data.get('content', '')
        code_blocks = data.get('codeBlocks', [])
        
        # If has code blocks, handle them
        if code_blocks:
            for block in code_blocks:
                self.handle_single_code(block, target_ide)
        else:
            # Just copy text
            pyperclip.copy(content)
            
            # Try to insert at cursor in active IDE
            if self.config.get('auto_insert', True):
                self.insert_text_in_ide(content, target_ide)
    
    def handle_all_code(self, data: Dict, target_ide: str):
        """Handle all code blocks"""
        blocks = data.get('blocks', [])
        
        if not blocks:
            return
        
        # Create combined file
        content = ""
        for i, block in enumerate(blocks):
            content += f"// Block {i + 1} ({block['language']})\n"
            content += block['code']
            content += "\n\n"

        temp_dir = Path.home() / '.deepseek-copier' / 'temp'
        temp_dir.mkdir(exist_ok=True)
        file_path = temp_dir / f"all-code-{int(time.time())}.txt"

        self.open_file_in_ide(file_path, target_ide)

class CopyHandler(BaseHTTPRequestHandler):
    server_instance = None
    
    def do_POST(self):
        content_length = int(self.headers['Content-Length'])
        post_data = self.rfile.read(content_length)
        
        try:
            data = json.loads(post_data.decode('utf-8'))
            result = self.server_instance.handle_copy(data)
            
            self.send_response(200)
            self.send_header('Content-Type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps(result).encode())
        except Exception as e:
            self.send_response(500)
            self.send_header('Content-Type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps({
                'status': 'error',
                'message': str(e)
            }).encode())
    
    def do_OPTIONS(self):
        self.send_response(200)
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()
    
    def do_GET(self):
        if self.path == '/status':
            self.send_response(200)
            self.send_header('Content-Type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            
            status = {
                'status': 'running',
                'ides': self.server_instance.ide_detector.list_available_ides(),
                'active_ide': self.server_instance.ide_detector.detect_active_ide(),
                'history_count': len(self.server_instance.history)
            }
            self.wfile.write(json.dumps(status).encode())
        elif self.path == '/config':
            self.send_response(200)
            self.send_header('Content-Type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps(self.server_instance.config).encode())
        else:
            self.send_response(404)
            self.end_headers()

import json
import hashlib
import difflib
import time
from pathlib import Path
from typing import Dict, List, Optional, Tuple
import re
from dataclasses import dataclass, asdict
from enum import Enum

class CodeStatus(Enum):
    """Status of code block"""
    ORIGINAL = "original"
    MODIFIED = "modified"
    ERROR = "error"
    FIXED = "fixed"
    OPTIMIZED = "optimized"
    REFACTORED = "refactored"
    DEPRECATED = "deprecated"

@dataclass
class CodeVersion:
    """Represents a version of a code block"""
    version_id: str
    code: str
    language: str
    timestamp: float
    status: CodeStatus
    parent_version: Optional[str] = None
    error_info: Optional[Dict] = None
    change_summary: Optional[str] = None
    hash: str = ""
    
    def __post_init__(self):
        if not self.hash:
            self.hash = hashlib.md5(self.code.encode()).hexdigest()
    
    def to_dict(self):
        return asdict(self)

@dataclass
class CodeBlock:
    """Represents a code block with version history"""
    block_id: str
    language: str
    versions: List[CodeVersion]
    current_version: str
    file_path: Optional[Path] = None
    context: Optional[Dict] = None
    created_at: float = time.time()
    updated_at: float = time.time()
    
    def get_current_code(self) -> str:
        """Get current version of code"""
        for version in self.versions:
            if version.version_id == self.current_version:
                return version.code
        return self.versions[-1].code if self.versions else ""
    
    def add_version(self, version: CodeVersion):
        """Add new version"""
        self.versions.append(version)
        self.current_version = version.version_id
        self.updated_at = time.time()
    
    def get_version(self, version_id: str) -> Optional[CodeVersion]:
        """Get specific version"""
        for version in self.versions:
            if version.version_id == version_id:
                return version
        return None
    
    def get_history(self) -> List[Dict]:
        """Get version history"""
        return [v.to_dict() for v in self.versions]
    
    def to_dict(self):
        return {
            'block_id': self.block_id,
            'language': self.language,
            'versions': [v.to_dict() for v in self.versions],
            'current_version': self.current_version,
            'file_path': str(self.file_path) if self.file_path else None,
            'context': self.context,
            'created_at': self.created_at,
            'updated_at': self.updated_at
        }

class CodeVersionManager:
    """Manages code versions and detects changes"""
    
    def __init__(self, storage_dir: Path):
        self.storage_dir = storage_dir
        self.storage_dir.mkdir(parents=True, exist_ok=True)
        self.blocks: Dict[str, CodeBlock] = {}
        self.load_state()
    
    def load_state(self):
        """Load saved state"""
        state_file = self.storage_dir / 'state.json'
        if state_file.exists():
            try:
                data = json.loads(state_file.read_text())
                for block_data in data.get('blocks', []):
                    block = self.block_from_dict(block_data)
                    self.blocks[block.block_id] = block
            except Exception as e:
                print(f"Error loading state: {e}")
    
    def save_state(self):
        """Save current state"""
        state_file = self.storage_dir / 'state.json'
        state = {
            'blocks': [block.to_dict() for block in self.blocks.values()]
        }
        state_file.write_text(json.dumps(state, indent=2))
    
    def block_from_dict(self, data: Dict) -> CodeBlock:
        """Create CodeBlock from dict"""
        versions = []
        for v_data in data['versions']:
            version = CodeVersion(
                version_id=v_data['version_id'],
                code=v_data['code'],
                language=v_data['language'],
                timestamp=v_data['timestamp'],
                status=CodeStatus(v_data['status']),
                parent_version=v_data.get('parent_version'),
                error_info=v_data.get('error_info'),
                change_summary=v_data.get('change_summary'),
                hash=v_data.get('hash', '')
            )
            versions.append(version)
        
        return CodeBlock(
            block_id=data['block_id'],
            language=data['language'],
            versions=versions,
            current_version=data['current_version'],
            file_path=Path(data['file_path']) if data.get('file_path') else None,
            context=data.get('context'),
            created_at=data.get('created_at', time.time()),
            updated_at=data.get('updated_at', time.time())
        )
    
    def identify_block(self, code: str, language: str, context: Optional[Dict] = None) -> Optional[str]:
        """Identify if code matches existing block"""
        code_hash = hashlib.md5(code.encode()).hexdigest()
        
        # Check for exact hash match
        for block_id, block in self.blocks.items():
            for version in block.versions:
                if version.hash == code_hash:
                    return block_id
        
        # Check for similarity (for slightly modified code)
        for block_id, block in self.blocks.items():
            if block.language != language:
                continue
            
            current_code = block.get_current_code()
            similarity = self.calculate_similarity(code, current_code)
            
            if similarity > 0.7:  # 70% similar
                return block_id
        
        return None
    
    def calculate_similarity(self, code1: str, code2: str) -> float:
        """Calculate similarity between two code strings"""
        if not code1 or not code2:
            return 0.0
        
        # Use difflib for similarity
        sequence_matcher = difflib.SequenceMatcher(None, code1, code2)
        return sequence_matcher.ratio()
    
    def detect_changes(self, old_code: str, new_code: str) -> Dict:
        """Detect and describe changes between code versions"""
        diff = list(difflib.unified_diff(
            old_code.splitlines(keepends=True),
            new_code.splitlines(keepends=True),
            fromfile='previous',
            tofile='current',
            n=3
        ))
        
        changes = {
            'diff': ''.join(diff),
            'added_lines': 0,
            'removed_lines': 0,
            'modified_lines': 0,
            'summary': []
        }
        
        for line in diff:
            if line.startswith('+') and not line.startswith('+++'):
                changes['added_lines'] += 1
            elif line.startswith('-') and not line.startswith('---'):
                changes['removed_lines'] += 1
        
        # Generate summary
        if changes['added_lines'] > 0 and changes['removed_lines'] > 0:
            changes['summary'].append(f"Modified {changes['added_lines'] + changes['removed_lines']} lines")
        elif changes['added_lines'] > 0:
            changes['summary'].append(f"Added {changes['added_lines']} lines")
        elif changes['removed_lines'] > 0:
            changes['summary'].append(f"Removed {changes['removed_lines']} lines")
        
        return changes
    
    def detect_error_fix(self, old_code: str, new_code: str) -> bool:
        """Detect if new code is fixing errors in old code"""
        # Look for common error patterns
        error_patterns = [
            r'Error:',
            r'Exception:',
            r'Traceback',
            r'TypeError',
            r'ValueError',
            r'AttributeError',
            r'SyntaxError',
            r'NameError',
            r'IndexError',
            r'KeyError',
            r'RuntimeError',
            r'ImportError'
        ]
        
        old_has_errors = any(re.search(pattern, old_code) for pattern in error_patterns)
        new_has_errors = any(re.search(pattern, new_code) for pattern in error_patterns)
        
        # If old code had errors and new code doesn't
        if old_has_errors and not new_has_errors:
            return True
        
        # Check for common fixes
        fix_patterns = [
            (r'except:', r'except Exception as e:'),  # Proper exception handling
            (r'print\(', r'print('),  # Print statement fixes
            (r'== None', r'is None'),  # None comparison
            (r'!= None', r'is not None'),  # None comparison
            (r'\.\.\.', 'pass'),  # Placeholder replacement
        ]
        
        for old_pattern, new_pattern in fix_patterns:
            if re.search(old_pattern, old_code) and re.search(new_pattern, new_code):
                return True
        
        return False
    
    def add_or_update_block(self, code: str, language: str, context: Optional[Dict] = None, 
                           error_info: Optional[Dict] = None) -> CodeBlock:
        """Add new block or update existing one"""
        block_id = self.identify_block(code, language, context)
        
        if block_id:
            # Update existing block
            block = self.blocks[block_id]
            current_code = block.get_current_code()
            
            # Detect changes
            changes = self.detect_changes(current_code, code)
            
            # Determine status
            if error_info:
                status = CodeStatus.ERROR
            elif self.detect_error_fix(current_code, code):
                status = CodeStatus.FIXED
                # If fixing errors, mark old version as error
                old_version = block.get_version(block.current_version)
                if old_version:
                    old_version.status = CodeStatus.ERROR
                    old_version.error_info = {
                        'message': 'Had errors, fixed in newer version',
                        'timestamp': time.time()
                    }
            elif changes['added_lines'] > 0 and changes['removed_lines'] == 0:
                status = CodeStatus.OPTIMIZED
            elif len(code) < len(current_code) * 0.7:
                status = CodeStatus.REFACTORED
            else:
                status = CodeStatus.MODIFIED
            
            # Create new version
            version = CodeVersion(
                version_id=f"v{len(block.versions) + 1}",
                code=code,
                language=language,
                timestamp=time.time(),
                status=status,
                parent_version=block.current_version,
                error_info=error_info,
                change_summary=', '.join(changes['summary'])
            )
            
            block.add_version(version)
            
            # Update file if exists
            if block.file_path and block.file_path.exists():
                block.file_path.write_text(code)
            
            return block
        else:
            # Create new block
            block_id = hashlib.md5(f"{language}-{time.time()}".encode()).hexdigest()[:12]
            
            version = CodeVersion(
                version_id="v1",
                code=code,
                language=language,
                timestamp=time.time(),
                status=CodeStatus.ORIGINAL if not error_info else CodeStatus.ERROR,
                error_info=error_info
            )
            
            block = CodeBlock(
                block_id=block_id,
                language=language,
                versions=[version],
                current_version="v1",
                context=context
            )
            
            self.blocks[block_id] = block
            return block
    
    def get_block(self, block_id: str) -> Optional[CodeBlock]:
        """Get block by ID"""
        return self.blocks.get(block_id)
    
    def get_all_blocks(self) -> List[CodeBlock]:
        """Get all blocks"""
        return list(self.blocks.values())
    
    def get_blocks_by_language(self, language: str) -> List[CodeBlock]:
        """Get blocks by language"""
        return [block for block in self.blocks.values() if block.language == language]
    
    def get_recent_blocks(self, limit: int = 10) -> List[CodeBlock]:
        """Get recently modified blocks"""
        sorted_blocks = sorted(
            self.blocks.values(),
            key=lambda b: b.updated_at,
            reverse=True
        )
        return sorted_blocks[:limit]
    
    def revert_to_version(self, block_id: str, version_id: str) -> bool:
        """Revert block to specific version"""
        block = self.get_block(block_id)
        if not block:
            return False
        
        version = block.get_version(version_id)
        if not version:
            return False
        
        # Create new version as revert
        new_version = CodeVersion(
            version_id=f"v{len(block.versions) + 1}",
            code=version.code,
            language=block.language,
            timestamp=time.time(),
            status=CodeStatus.REFACTORED,
            parent_version=block.current_version,
            change_summary=f"Reverted to {version_id}"
        )
        
        block.add_version(new_version)
        
        # Update file
        if block.file_path and block.file_path.exists():
            block.file_path.write_text(version.code)
        
        return True
    
    def cleanup_old_versions(self, max_versions_per_block: int = 10):
        """Clean up old versions to save space"""
        for block in self.blocks.values():
            if len(block.versions) > max_versions_per_block:
                # Keep the newest versions and the original
                kept_versions = block.versions[-max_versions_per_block:]
                block.versions = kept_versions
                block.current_version = kept_versions[-1].version_id
    
    def export_history(self, format: str = 'json') -> Path:
        """Export version history"""
        export_file = self.storage_dir / f'history_{int(time.time())}.{format}'
        
        if format == 'json':
            data = {
                'blocks': [block.to_dict() for block in self.blocks.values()]
            }
            export_file.write_text(json.dumps(data, indent=2))
        elif format == 'markdown':
            markdown = "# Code Version History\n\n"
            
            for block in self.blocks.values():
                markdown += f"## Block {block.block_id} ({block.language})\n\n"
                
                for version in block.versions:
                    markdown += f"### {version.version_id} - {version.status.value}\n"
                    markdown += f"**Time:** {time.strftime('%Y-%m-%d %H:%M:%S', time.localtime(version.timestamp))}\n"
                    if version.change_summary:
                        markdown += f"**Changes:** {version.change_summary}\n"
                    if version.error_info:
                        markdown += f"**Errors:** {version.error_info.get('message', 'Unknown')}\n"
                    markdown += f"\n```{block.language}\n{version.code}\n```\n\n"
            
            export_file.write_text(markdown)
        
        return export_file

class ErrorTracker:
    """Tracks and analyzes code errors"""
    
    def __init__(self):
        self.error_history = []
        self.error_patterns = {}
    
    def track_error(self, code: str, error_message: str, language: str):
        """Track an error occurrence"""
        error_entry = {
            'timestamp': time.time(),
            'language': language,
            'error_message': error_message,
            'code_snippet': code[:500]  # Store first 500 chars
        }
        self.error_history.append(error_entry)
        
        # Track error patterns
        error_type = self.classify_error(error_message)
        if error_type not in self.error_patterns:
            self.error_patterns[error_type] = {
                'count': 0,
                'examples': [],
                'fixes': []
            }
        
        self.error_patterns[error_type]['count'] += 1
        if len(self.error_patterns[error_type]['examples']) < 5:
            self.error_patterns[error_type]['examples'].append({
                'error': error_message,
                'code': code[:200]
            })
    
    def classify_error(self, error_message: str) -> str:
        """Classify error type"""
        error_types = {
            'syntax_error': r'SyntaxError|syntax error',
            'type_error': r'TypeError|type error',
            'value_error': r'ValueError|value error',
            'attribute_error': r'AttributeError|attribute error',
            'name_error': r'NameError|name error|undefined',
            'index_error': r'IndexError|index error',
            'key_error': r'KeyError|key error',
            'import_error': r'ImportError|module not found|import error',
            'runtime_error': r'RuntimeError|runtime error',
            'indentation_error': r'IndentationError|indentation error',
            'zero_division': r'ZeroDivisionError|division by zero',
            'file_not_found': r'FileNotFoundError|no such file',
            'permission_error': r'PermissionError|permission denied'
        }
        
        for error_type, pattern in error_types.items():
            if re.search(pattern, error_message, re.IGNORECASE):
                return error_type
        
        return 'unknown_error'

            'value_error': "Check if the value is within acceptable range",
            'attribute_error': "Verify that the attribute exists on the object",

class VersionManager {
    constructor() {
        this.versions = new Map();
        this.loadState();
    }
    
    loadState() {
        chrome.storage.local.get(['codeVersions'], (result) => {
            if (result.codeVersions) {
                this.versions = new Map(Object.entries(result.codeVersions));
            }
        });
    }
    
    saveState() {
        const data = Object.fromEntries(this.versions);
        chrome.storage.local.set({codeVersions: data});
    }
    
    trackCode(codeBlock, errorInfo = null) {
        const hash = this.hashCode(codeBlock.code);
        const existing = this.versions.get(hash);
        
        if (existing) {
            // Update version
            existing.versions.push({
                timestamp: Date.now(),
                code: codeBlock.code,
                errorInfo: errorInfo,
                version: existing.versions.length + 1
            });
        } else {
            // New code block
            this.versions.set(hash, {
                blockId: hash,
                language: codeBlock.language,
                versions: [{
                    timestamp: Date.now(),
                    code: codeBlock.code,
                    errorInfo: errorInfo,
                    version: 1
                }]
            });
        }
        
        this.saveState();
    }
    
    hashCode(str) {
        let hash = 0;
        for (let i = 0; i < str.length; i++) {
            const char = str.charCodeAt(i);
            hash = ((hash << 5) - hash) + char;
            hash = hash & hash;
        }
        return hash.toString(36);
    }
}

class ErrorDetector {
    detectError(text) {
        if (!text) return null;
        
        const errorPatterns = [
            /Error:/,
            /Exception:/,
            /Traceback/,
            /TypeError/,
            /ValueError/,
            /AttributeError/,
            /SyntaxError/,
            /NameError/,
            /IndexError/,
            /KeyError/,
            /RuntimeError/,
            /ImportError/,
            /not working/,
            /doesn't work/,
            /failed/,
            /bug/,
            /issue/
        ];
        
        for (const pattern of errorPatterns) {
            if (pattern.test(text)) {
                return {
                    message: this.extractErrorMessage(text),
                    pattern: pattern.source,
                    timestamp: Date.now()
                };
            }
        }
        
        return null;
    }
    
    extractErrorMessage(text) {
        const patterns = [
            /(?:Error|Exception|Traceback)[:\s]+(.+?)(?:\n|$)/i,
            /(?:error|exception)[:\s]+(.+?)(?:\n|$)/i,
            /Failed[:\s]+(.+?)(?:\n|$)/i
        ];
        
        for (const pattern of patterns) {
            const match = text.match(pattern);
            if (match) {
                return match[1].trim();
            }
        }
        
        return "Unknown error";
    }
}

        const oldLines = oldVersion.code.split('\n');
        const newLines = newVersion.code.split('\n');
        
        let diff = '';
        const maxLines = Math.max(oldLines.length, newLines.length);
        
        for (let i = 0; i < maxLines; i++) {
            if (i < oldLines.length && i < newLines.length) {
                if (oldLines[i] !== newLines[i]) {
                    diff += `<div class="removed">- ${oldLines[i]}</div>`;
                    diff += `<div class="added">+ ${newLines[i]}</div>`;
                } else {
                    diff += `<div>  ${oldLines[i]}</div>`;
                }
            } else if (i < oldLines.length) {
                diff += `<div class="removed">- ${oldLines[i]}</div>`;
            } else if (i < newLines.length) {
                diff += `<div class="added">+ ${newLines[i]}</div>`;
            }
        }
        
        return `<div class="diff-view">${diff}</div>`;
    }

import asyncio

from typing import Dict, Set

class WebSocketServer:
    def __init__(self):
        self.clients: Set[websockets.WebSocketServerProtocol] = set()
        self.message_queue = asyncio.Queue()
    
    async def register(self, websocket):
        self.clients.add(websocket)
        try:
            async for message in websocket:
                await self.handle_message(websocket, message)
        finally:
            self.clients.remove(websocket)
    
    async def handle_message(self, websocket, message):
        data = json.loads(message)
        # Process message
        response = await self.process_data(data)
        # Broadcast to all clients
        await self.broadcast(response)
    
    async def broadcast(self, message):
        if self.clients:
            await asyncio.gather(
                *[client.send(json.dumps(message)) for client in self.clients]
            )

import ast
import esprima  # For JavaScript
from typing import Dict, List, Optional

class CodeAnalyzer:
    def analyze_python_code(self, code: str) -> Dict:
        """Analyze Python code structure"""
        try:
            tree = ast.parse(code)
            return {
                'imports': self.extract_imports(tree),
                'functions': self.extract_functions(tree),
                'classes': self.extract_classes(tree),
                'complexity': self.calculate_complexity(tree),
                'dependencies': self.find_dependencies(tree)
            }
        except SyntaxError as e:
            return {'error': str(e)}
    
    def extract_imports(self, tree):
        imports = []
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                imports.extend(alias.name for alias in node.names)
            elif isinstance(node, ast.ImportFrom):
                imports.append(node.module)
        return imports
    
    def extract_functions(self, tree):
        functions = []
        for node in ast.walk(tree):
            if isinstance(node, ast.FunctionDef):
                functions.append({
                    'name': node.name,
                    'args': [arg.arg for arg in node.args.args],
                    'lineno': node.lineno,
                    'decorators': [d.id for d in node.decorator_list]
                })
        return functions
    
    def calculate_complexity(self, tree):
        """Calculate cyclomatic complexity"""
        complexity = 1
        for node in ast.walk(tree):
            if isinstance(node, (ast.If, ast.While, ast.For, ast.ExceptHandler)):
                complexity += 1
            elif isinstance(node, ast.BoolOp):
                complexity += len(node.values) - 1
        return complexity

    def predict_error_probability(self, code):
        """Predict if code likely has errors"""
        try:
            vectorizer = joblib.load(self.model_path / 'vectorizer.pkl')
            classifier = joblib.load(self.model_path / 'classifier.pkl')
            
            X = vectorizer.transform([code])
            proba = classifier.predict_proba(X)[0]
            return {
                'error_probability': proba[1],
                'confidence': max(proba)
            }

from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
import base64

class SecureStorage:
    def __init__(self, password=None):
        self.key = self.generate_key(password)
        self.cipher = Fernet(self.key)
    
    def generate_key(self, password):
        if not password:
            # Generate random key
            return Fernet.generate_key()
        
        # Derive key from password
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=b'deepseek-copier',
            iterations=100000,
        )
        key = base64.urlsafe_b64encode(kdf.derive(password.encode()))
        return key
    
    def encrypt_code(self, code):
        return self.cipher.encrypt(code.encode())
    
    def decrypt_code(self, encrypted_code):
        return self.cipher.decrypt(encrypted_code).decode()

class BatchProcessor:
    def __init__(self, batch_size=10):
        self.batch_size = batch_size
        self.pending_items = []
        self.processing = False
    
    async def add_item(self, item):
        self.pending_items.append(item)
        if len(self.pending_items) >= self.batch_size:
            await self.process_batch()
    
    async def process_batch(self):
        if self.processing:
            return
        
        self.processing = True
        batch = self.pending_items[:self.batch_size]
        self.pending_items = self.pending_items[self.batch_size:]
        
        # Process batch in parallel
        results = await asyncio.gather(
            *[self.process_item(item) for item in batch]
        )
        
        self.processing = False
        return results

-- Code Blocks Table
CREATE TABLE code_blocks (
    id TEXT PRIMARY KEY,
    language TEXT NOT NULL,
    current_version TEXT,
    file_path TEXT,
    context JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Code Versions Table
CREATE TABLE code_versions (
    id TEXT PRIMARY KEY,
    block_id TEXT REFERENCES code_blocks(id),
    version_number INTEGER,
    code TEXT NOT NULL,
    hash TEXT NOT NULL,
    status TEXT,
    parent_version TEXT,
    error_info JSON,
    change_summary TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Error Tracking Table
CREATE TABLE error_tracking (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    version_id TEXT REFERENCES code_versions(id),
    error_type TEXT,
    error_message TEXT,
    code_snippet TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- IDE Configurations Table
CREATE TABLE ide_configurations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    ide_type TEXT,
    ide_name TEXT,
    command TEXT,
    is_active BOOLEAN DEFAULT FALSE,
    config JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- User Preferences Table
CREATE TABLE user_preferences (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    key TEXT UNIQUE,
    value JSON,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

❓ FAQ
General
Q: Is my code stored securely?
A: Yes, all code is stored locally. You can enable encryption for additional security.

Q: Does it work offline?
A: Yes, the system works entirely locally.

Q: Can I use multiple IDEs?
A: Yes, the system supports multiple IDEs and can switch between them.

Q: Is it free?
A: Yes, the basic version is open source and free.

Technical
Q: What languages are supported?
A: All languages that DeepSeek supports, with special handling for 20+ popular languages.

Q: How does version tracking work?
A: Each code change creates a new version with diff tracking and change summaries.

Q: Can I customize the detection?
A: Yes, you can customize code block detection patterns in settings.

Q: Does it work with Jupyter notebooks?
A: Yes, it can export code to Jupyter notebooks.

Troubleshooting
Q: Why isn't my IDE detected?
A: Ensure the IDE is in your PATH or add it manually in settings.

Q: Can I recover deleted code?
A: Yes, if version tracking is enabled, you can recover previous versions.

Q: How do I export my data?
A: Use the export feature in the dashboard or CLI.

Q: Is there a size limit?
